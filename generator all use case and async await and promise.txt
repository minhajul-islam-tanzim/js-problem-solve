Object.prototype[Symbol.iterator] = function(){
   const ent = Object.entries(this)
    let count = ent.length;

    let ind = 0;

    return {
        next(){
            if(count > 0){
                let result = {done : false , value: ent[ind][1]};
                count--
                ind++
                return result;

            }
                return {done: true}

          }
     }
}

const pro = (cus) => {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve(`order processed for ${cus}`)
        }, 2000)

    })
}
const com = (cus) => {
    return new Promise((resolve) => {
        setTimeout(() => {
           resolve(`order complated for ${cus}`)
        }, 3000)

    })
}

async function *sol(cus){
    yield await pro(cus)
    yield await com(cus)
}

const gen = sol('karim')

gen.next().then(({value})=> console.log(value))

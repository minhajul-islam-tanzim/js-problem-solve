let w;


function stW(){

    if(typeof Worker !== "undefined"){

            if(typeof w == 'undefined'){

                w = new Worker('./js/worker.js') // সঠিক path

            }

            w.onmessage = function (event){
                const demo = document.getElementById('demo')
                demo.innerHTML = event.data;
            }


    }else{

    alert('Worker is not support in this browser')

    }

}



function stopW(){
    if(typeof w !== 'undefined'){
        w.terminate();
        w = undefined;
    }else{
        alert('worker naiii bapp')

    }
}

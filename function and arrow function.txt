const cars = [
    {
        name: "BMW",
        model: "W1",
        used: "3y",
        price:"100"
    },
    {
        name: "BMW",
        model: "W2",
        used: "3y",
        price:"101"
    },
    {
        name: "BMW",
        model: "W3",
        used: "3y",
        price:"99"
    }
];

let buy = (cars) => {
  return cars.filter((cars) => cars.price >= 100).map((x) => x.name)
    
}

let x = buy(cars)
console.log(x)

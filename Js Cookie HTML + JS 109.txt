function setName(cname, cvalue, ex) {
    let date = new Date();
    date.setTime(date.getTime() + 24 * 60 * 60 * ex * 1000)
    let expires = 'expires=' + date.toUTCString();

    document.cookie = cname + '=' + cvalue + ';' + expires + ';path=/';

    console.log(date)

}


function getCookie2(cname) {

    let allName = cname + '=';
    let ca = document.cookie.split(';')
// ['name=tanzim ', 'username=tanzim888']




    for (let i = 0; i < ca.length; i++) {
        let res = ca[i]


        while (res.charAt(0) === " ") {
            res = res.substring(1)
        }

        if (res.indexOf(allName) === 0) {
            return res.substring(allName.length)
        }


    }

    return '';


}

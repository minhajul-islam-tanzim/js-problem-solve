js>>>>>>>>>>>>>>>>>>>>>>>>>>

const sq = document.querySelectorAll('div');

const obs = new IntersectionObserver((sq) => {

    sq.forEach(squ => {
        if(squ.isIntersecting){
            squ.target.classList.add('vis')

        }else{
            squ.target.classList.remove('vis')
        }

    })

})

  sq.forEach(squ => obs.observe(squ))
